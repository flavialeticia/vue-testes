<template>
  <div id="app">
    <h1>{{ mensagem }}</h1>

    <p>Contador: {{ contador }}</p>

    <!-- Usando valores computados no template -->
    <p>Contador em dobro (computed): {{ contadorDobrado }}</p>
    <p>Status (par/ímpar): {{ statusParidade }}</p>

    <div class="button-container">
      <button @click="decrementar">-1</button>
      <button @click="incrementar">+1</button>
    </div>
  </div>
</template>

<script setup>
import { ref, computed } from 'vue'

// Estado base
const mensagem = ref('Hello, Vue!')
const contador = ref(0)

// Ações
function incrementar() {
  contador.value++
}

function decrementar() {
  contador.value--
}

// Valores derivados (computed) — cacheados
const contadorDobrado = computed(() => contador.value * 2)
const statusParidade  = computed(() => (contador.value % 2 === 0 ? 'par' : 'ímpar'))

/*
- computed(() => ...) define um valor derivado com cache.
- Só recalcula quando 'contador.value' mudar.
*/
</script>
