<template>
  <div id="app">
    <h1>{{ mensagem }}</h1>

    <!-- Primeiro Contador -->
    <div class="contador-section">
      <h2>Contador A</h2>
      <p>Contador: {{ contadorA }}</p>
      <p>Contador em dobro (computed): {{ contadorADobrado }}</p>
      <p>Status (par/ímpar): {{ statusParidadeA }}</p>
      
      <div class="button-container">
        <button @click="decrementarA">-1</button>
        <button @click="incrementarA">+1</button>
      </div>
    </div>

    <!-- Segundo Contador -->
    <div class="contador-section">
      <h2>Contador B</h2>
      <p>Contador: {{ contadorB }}</p>
      <p>Contador em dobro (computed): {{ contadorBDobrado }}</p>
      <p>Status (par/ímpar): {{ statusParidadeB }}</p>
      
      <div class="button-container">
        <button @click="decrementarB">-1</button>
        <button @click="incrementarB">+1</button>
      </div>
    </div>

    <!-- Soma dos Contadores -->
    <div class="soma-section">
      <h2>Soma dos Contadores</h2>
      <p class="soma-resultado">Soma: {{ somaContadores }}</p>
      <p class="soma-dobro">Soma em dobro: {{ somaDobro }}</p>
      <p class="soma-status">Status da soma: {{ statusSoma }}</p>
    </div>
  </div>
</template>

<script setup>
import { ref, computed } from 'vue'

// Estado base
const mensagem = ref('Hello, Vue!')
const contadorA = ref(0)
const contadorB = ref(0)

// Ações para o Contador A
function incrementarA() {
  contadorA.value++
}

function decrementarA() {
  contadorA.value--
}

// Ações para o Contador B
function incrementarB() {
  contadorB.value++
}

function decrementarB() {
  contadorB.value--
}

// Valores derivados (computed) para Contador A
const contadorADobrado = computed(() => contadorA.value * 2)
const statusParidadeA = computed(() => (contadorA.value % 2 === 0 ? 'par' : 'ímpar'))

// Valores derivados (computed) para Contador B
const contadorBDobrado = computed(() => contadorB.value * 2)
const statusParidadeB = computed(() => (contadorB.value % 2 === 0 ? 'par' : 'ímpar'))

// Soma dos contadores
const somaContadores = computed(() => contadorA.value + contadorB.value)
const somaDobro = computed(() => somaContadores.value * 2)
const statusSoma = computed(() => (somaContadores.value % 2 === 0 ? 'par' : 'ímpar'))
</script>
