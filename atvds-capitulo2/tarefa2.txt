<template>
  <form @submit.prevent="adicionar">
    <input v-model.trim="novoTexto" placeholder="Digite a mensagem" />
    <button>Adicionar</button>
  </form>

  <button v-if="estado.itens.length" @click="limpar">Limpar Tudo</button>

  <ul v-if="estado.itens.length > 0">
    <li v-for="item in estado.itens" :key="item.id">
      #{{ item.id }} — {{ item.texto }}
      <button @click="remover(item.id)">Remover</button>
    </li>
  </ul>
  <p v-else>Lista vazia.</p>
</template>

<script setup>
import { reactive, ref } from 'vue'

const estado = reactive({
  nextId: 1,
  itens: []
})

const novoTexto = ref('')

function adicionar() {
  if (novoTexto.value.trim() === '') return
  estado.itens.push({
    id: estado.nextId,
    texto: novoTexto.value
  })
  estado.nextId++
  novoTexto.value = ''
}

function remover(id) {
  estado.itens = estado.itens.filter(i => i.id !== id)
}

function limpar() {
  estado.itens = []
}
</script>
