<template>
  <form @submit.prevent="adicionar">
    <input v-model.trim="novoTexto" placeholder="Digite a mensagem" />
    <button :disabled="!novoTexto">Adicionar</button>
    <p v-if="erro" style="color: red; font-size: 0.9rem;">{{ erro }}</p>
  </form>

  <div v-if="estado.itens.length > 0" style="margin-top: 1rem;">
    <input v-model="filtro" placeholder="Filtrar mensagens..." />

    <p>Exibindo {{ filtradas.length }} de {{ estado.itens.length }}</p>
    <button @click="limpar">Limpar Tudo</button>

    <ul>
      <li
        v-for="item in filtradas"
        :key="item.id"
        :class="{ longo: item.texto.length > 30 }"
      >
        #{{ item.id }} — {{ item.texto }}
        <button @click="remover(item.id)">Remover</button>
      </li>
    </ul>

    <p v-if="!filtradas.length">Nenhuma mensagem encontrada.</p>
  </div>

  <p v-else>Lista vazia.</p>
</template>

<script setup>
import { reactive, ref, computed } from 'vue'

const estado = reactive({
  nextId: 1,
  itens: []
})

const novoTexto = ref('')
const filtro = ref('')
const erro = ref('')

function adicionar() {
  if (novoTexto.value.trim() === '') {
    erro.value = '⚠️ Digite uma mensagem antes de adicionar.'
    return
  }

  estado.itens.push({
    id: estado.nextId,
    texto: novoTexto.value
  })
  estado.nextId++
  novoTexto.value = ''
  erro.value = ''
}

function remover(id) {
  estado.itens = estado.itens.filter(i => i.id !== id)
}

function limpar() {
  estado.itens = []
}

// Computed: filtra mensagens conforme o texto digitado
const filtradas = computed(() => {
  const termo = filtro.value.toLowerCase()
  return estado.itens.filter(i => i.texto.toLowerCase().includes(termo))
})
</script>

<style scoped>
.longo {
  background-color: #fff3cd;
  border-left: 4px solid #ffb300;
  padding: 4px 8px;
  border-radius: 4px;
}
button:disabled {
  background-color: #ccc;
  cursor: not-allowed;
}
</style>
