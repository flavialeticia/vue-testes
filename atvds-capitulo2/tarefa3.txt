<template>
  <form @submit.prevent="adicionar">
    <input v-model.trim="novoTexto" placeholder="Digite a mensagem" />
    <button>Adicionar</button>
  </form>

  <div v-if="estado.itens.length > 0" style="margin-top: 1rem;">
    <input v-model="filtro" placeholder="Filtrar mensagens..." />

    <p>Exibindo {{ filtradas.length }} de {{ estado.itens.length }}</p>
    <button @click="limpar">Limpar Tudo</button>

    <ul>
      <li v-for="item in filtradas" :key="item.id">
        #{{ item.id }} — {{ item.texto }}
        <button @click="remover(item.id)">Remover</button>
      </li>
    </ul>
  </div>

  <p v-else>Lista vazia.</p>
</template>

<script setup>
import { reactive, ref, computed } from 'vue'

const estado = reactive({
  nextId: 1,
  itens: []
})

const novoTexto = ref('')
const filtro = ref('')

function adicionar() {
  if (novoTexto.value.trim() === '') return
  estado.itens.push({
    id: estado.nextId,
    texto: novoTexto.value
  })
  estado.nextId++
  novoTexto.value = ''
}

function remover(id) {
  estado.itens = estado.itens.filter(i => i.id !== id)
}

function limpar() {
  estado.itens = []
}

// Computed: filtra mensagens conforme o texto digitado em "filtro"
const filtradas = computed(() => {
  const termo = filtro.value.toLowerCase()
  return estado.itens.filter(i => i.texto.toLowerCase().includes(termo))
})
</script>
